def rj(path, default):
    p = Path(path)
    if p.exists():
        try:
            return json.loads(p.read_text(encoding="utf-8"))
        except Exception:
            return default
    return default


def wj(path, data):
    Path(path).write_text(json.dumps(data, indent=2, ensure_ascii=False), encoding="utf-8")

def think(prompt, conversation_history=None, system=None):
    if conversation_history is None:
        conversation_history = []
    try:
        messages = list(conversation_history) + [{"role": "user", "content": prompt}]
        r = requests.post(
            "https://api.anthropic.com/v1/messages",
            headers={
                "x-api-key": ANTHROPIC_API_KEY,
                "anthropic-version": "2023-06-01",
                "content-type": "application/json"
            },
            json={
                "model": "claude-sonnet-4-6",
                "max_tokens": 1500,
                "system": system or NEXUS_SOUL,
                "messages": messages
            },
            timeout=30
        )
        return r.json()["content"][0]["text"]
    except Exception as e:
        return "Nexus error: " + str(e)


def think_json(prompt):
    system = NEXUS_SOUL + "\nIMPORTANT: Respond ONLY with valid JSON. No markdown, no explanation."
    raw = think(prompt, system=system)
    raw = raw.strip()
    if raw.startswith("```"):
        parts = raw.split("```")
        raw = parts[1] if len(parts) > 1 else raw
        if raw.startswith("json"):
            raw = raw[4:]
    try:
        return json.loads(raw.strip())
    except Exception:
        return None

def get_price(symbol):
    try:
        token = symbol.replace("/USDT", "").replace("/USD", "").upper()
        r = requests.get(
            "https://api.dexscreener.com/latest/dex/search?q=" + token,
            timeout=10
        )
        pairs = r.json().get("pairs", [])
        sol = [p for p in pairs if p.get("chainId") == "solana"]
        if not sol:
            sol = pairs
        if not sol:
            return {"symbol": symbol, "price": None}
        best = max(sol, key=lambda p: float((p.get("liquidity") or {}).get("usd", 0) or 0))
        return {
            "symbol": symbol,
            "price": float(best.get("priceUsd", 0) or 0),
            "change_24h": float((best.get("priceChange") or {}).get("h24", 0) or 0),
            "volume_24h": float((best.get("volume") or {}).get("h24", 0) or 0),
            "liquidity": float((best.get("liquidity") or {}).get("usd", 0) or 0),
            "dex": best.get("dexId", ""),
            "pair": best.get("pairAddress", "")
        }
    except Exception as e:
        return {"symbol": symbol, "price": None, "error": str(e)}


def search_dex(query):
    try:
        r = requests.get(
            "https://api.dexscreener.com/latest/dex/search?q=" + query,
            timeout=10
        )
        pairs = r.json().get("pairs", [])
        if not pairs:
            return "No pairs found."
        top = sorted(pairs, key=lambda p: float((p.get("volume") or {}).get("h24", 0) or 0), reverse=True)[:5]
        lines = []
        for p in top:
            base = (p.get("baseToken") or {}).get("symbol", "?")
            quote = (p.get("quoteToken") or {}).get("symbol", "?")
            price = float(p.get("priceUsd", 0) or 0)
            change = float((p.get("priceChange") or {}).get("h24", 0) or 0)
            vol = float((p.get("volume") or {}).get("h24", 0) or 0)
            chain = p.get("chainId", "?")
            lines.append(
                base + "/" + quote + " on " + chain +
                " | $" + str(round(price, 8)) +
                " | 24h: " + str(round(change, 1)) + "%" +
                " | Vol: $" + str(int(vol))
            )
        return "\n".join(lines)
    except Exception as e:
        return "DexScreener error: " + str(e)

class TradingEngine:
    def __init__(self):
        self.positions = load_positions()
        self.pubkey = self._get_pubkey()
        mode = "PAPER" if PAPER_TRADING else "LIVE"
        print("Solana wallet: " + (self.pubkey or "not initialized") + " (" + mode + ")")

    def _get_pubkey(self):
        try:
            import base58
            raw = base58.b58decode(SOLANA_PRIVATE_KEY)
            if len(raw) == 64:
                return base58.b58encode(raw[32:]).decode()
        except Exception:
            pass
        return "wallet_configured"

    def get_balance(self):
        try:
            if not self.pubkey or self.pubkey == "wallet_configured":
                return {"note": "install base58 to read live balance"}
            r = requests.post(
                "https://api.mainnet-beta.solana.com",
                json={"jsonrpc": "2.0", "id": 1, "method": "getBalance", "params": [self.pubkey]},
                timeout=10
            )
            lamports = ((r.json().get("result") or {}).get("value") or 0)
            return {"SOL": round(lamports / 1e9, 4)}
        except Exception as e:
            return {"error": str(e)}

    def get_jupiter_quote(self, input_mint, output_mint, amount):
        try:
            r = requests.get(
                "https://quote-api.jup.ag/v6/quote",
                params={
                    "inputMint": input_mint,
                    "outputMint": output_mint,
                    "amount": str(int(amount)),
                    "slippageBps": "50"
                },
                timeout=10
            )
            return r.json()
        except Exception as e:
            return {"error": str(e)}

    def execute_trade(self, symbol, side, amount_usdt):
        token = symbol.replace("/USDT", "").replace("/USD", "").upper()
        pd = get_price(token)
        price = pd.get("price", 0)
        if not price:
            return {"success": False, "error": "No price data"}

        amount = min(float(amount_usdt), MAX_TRADE_USDT)
        input_mint = USDC_MINT if side == "buy" else TOKEN_MINTS.get(token, SOL_MINT)
        output_mint = TOKEN_MINTS.get(token, SOL_MINT) if side == "buy" else USDC_MINT

        if PAPER_TRADING:
            trade = {
                "symbol": symbol, "side": side,
                "amount_usdt": amount, "price": price,
                "paper": True, "id": "paper_" + str(int(time.time()))
            }
        else:
            quote = self.get_jupiter_quote(input_mint, output_mint, int(amount * 1e6))
            if "error" in quote:
                return {"success": False, "error": quote["error"]}
            trade = {
                "symbol": symbol, "side": side,
                "amount_usdt": amount, "price": price,
                "paper": False, "out_amount": quote.get("outAmount"),
                "id": "jup_" + str(int(time.time()))
            }

        if side == "buy":
            self.positions[symbol] = {
                "entry_price": price,
                "amount_usdt": amount,
                "entry_time": datetime.now().isoformat()
            }
        elif side == "sell" and symbol in self.positions:
            del self.positions[symbol]

        save_positions(self.positions)
        log_trade(trade)
        return {"success": True, "trade": trade}

    def check_stops(self):
        alerts = []
        for symbol, pos in list(self.positions.items()):
            token = symbol.replace("/USDT", "")
            pd = get_price(token)
            if pd.get("price"):
                current = pd["price"]
                entry = pos["entry_price"]
                pnl = ((current - entry) / entry) * 100
                if pnl <= -5.0:
                    self.execute_trade(symbol, "sell", pos["amount_usdt"])
                    alerts.append("STOP LOSS " + symbol + ": " + str(round(pnl, 1)) + "%")
                elif pnl >= 15.0:
                    self.execute_trade(symbol, "sell", pos["amount_usdt"])
                    alerts.append("TAKE PROFIT " + symbol + ": +" + str(round(pnl, 1)) + "%")
        return alerts

    print("[" + datetime.now().strftime("%H:%M") + "] Whale tracker...")
    alerts = []
    try:
        r = requests.get(
            "https://public-api.solscan.io/transaction/last",
            params={"limit": 20, "type": "sol_transfer"},
            headers={"accept": "application/json"},
            timeout=10
        )
        txs = r.json() if isinstance(r.json(), list) else []
        for tx in txs:
            lamports = tx.get("lamport", 0) or 0
            sol_amount = lamports / 1e9
            if sol_amount >= 1000:
                signer = ((tx.get("signer") or ["?"])[0])[:12]
                alerts.append("WHALE: " + str(int(sol_amount)) + " SOL from " + signer + "...")
    except Exception as e:
        print("[Whale] Solscan error: " + str(e))

    for wallet in WATCH_WALLETS:
        try:
            r = requests.get(
                "https://public-api.solscan.io/account/transactions",
                params={"account": wallet, "limit": 5},
                headers={"accept": "application/json"},
                timeout=10
            )
            txs = r.json() if isinstance(r.json(), list) else []
            if txs:
                alerts.append("WATCH WALLET " + wallet[:8] + "... active: " + str(len(txs)) + " recent txs")
        except Exception:
            pass

    if not alerts:
        print("[Whale] No significant activity")
        return

    alert_text = "\n".join(alerts)
    analysis = think(
        "[WHALE ACTIVITY]\n" + alert_text + "\n\n"
        "What does this signal for the market? Trading opportunity? Be specific.",
        conversation_history=[]
    )
    whale_log = rj(WHALE_FILE, [])
    whale_log.insert(0, {"time": datetime.now().isoformat(), "alerts": alerts, "analysis": analysis})
    wj(WHALE_FILE, whale_log[:50])
    send_telegram("[Nexus Whale Tracker]\n\n" + alert_text + "\n\n" + analysis[:500])
    print("[Whale] " + str(len(alerts)) + " alerts sent")

def handle_message(text, photo=None, document=None, caption=""):
    history = load_conv()

    # Image
    if photo:
        data, mime = download_file(photo[-1]["file_id"])
        if data:
            reply = think_with_image(base64.b64encode(data).decode(), mime, caption or "", history[-10:])
            history.append({"role": "user", "content": "[Image] " + caption})
            history.append({"role": "assistant", "content": reply})
            save_conv(history)
        else:
            reply = "Could not process image."
        return reply

    # Document
    if document:
        fname = document.get("file_name", "file")
        data, mime = download_file(document["file_id"])
        if data:
            try:
                file_text = data.decode("utf-8")
                reply = think("File: " + fname + "\n\n" + file_text[:4000] + "\n\n" + (caption or "Analyze this."), history[-10:])
            except Exception:
                if mime == "application/pdf":
                    try:
                        pdf_b64 = base64.b64encode(data).decode()
                        msgs = list(history[-10:]) + [{"role": "user", "content": [
                            {"type": "document", "source": {"type": "base64", "media_type": "application/pdf", "data": pdf_b64}},
                            {"type": "text", "text": caption or "Analyze this PDF."}
                        ]}]
                        r = requests.post(
                            "https://api.anthropic.com/v1/messages",
                            headers={"x-api-key": ANTHROPIC_API_KEY, "anthropic-version": "2023-06-01", "content-type": "application/json"},
                            json={"model": "claude-sonnet-4-6", "max_tokens": 1500, "system": NEXUS_SOUL, "messages": msgs},
                            timeout=30
                        )
                        reply = r.json()["content"][0]["text"]
                    except Exception as e:
                        reply = "Could not read PDF: " + str(e)
                else:
                    reply = "Cannot read binary file '" + fname + "'."
            history.append({"role": "user", "content": "[File: " + fname + "] " + caption})
            history.append({"role": "assistant", "content": reply})
            save_conv(history)
        else:
            reply = "Could not download file."
        return reply

    t = (text or "").strip()
    tl = t.lower()

    if tl == "/start":
        reply = (
            "Nexus online. Running autonomously.\n\n"
            "Active loops:\n"
            "Market scan - every 15 min\n"
            "DEX discovery - every 20 min\n"
            "Whale tracker - every 30 min\n"
            "Learning - every 1 hour\n"
            "Research - every 2 hours\n"
            "Sentiment - every 2 hours\n"
            "Content - every 3 hours\n"
            "Deep thinking - every 6 hours\n\n"
            "I'm always here."
        )

    elif tl == "/status":
        trades = load_trades()
        strategy = load_strategy()
        market = rj(MARKET_FILE, {})
        reply = (
            "NEXUS STATUS [" + datetime.now().strftime("%H:%M") + "]\n\n"
            "Trades: " + str(len(trades)) + " | Win rate: " + str(round(strategy.get("win_rate", 0) * 100, 1)) + "%\n"
            "Strategy: v" + str(strategy.get("version", 1)) + "\n"
            "Positions: " + str(len(trader.positions)) + " open\n"
            "  " + (", ".join(trader.positions.keys()) or "none") + "\n"
            "Last scan: " + market.get("time", "never")[:16] + "\n"
            "Market: " + market.get("summary", "none")[:150]
        )

    elif tl == "/market":
        prices = [get_price(s) for s in CRYPTO_WATCHLIST]
        poly = get_polymarket()
        price_lines = [
            s + ": $" + str(prices[i].get("price", 0)) + " (" + str(round(prices[i].get("change_24h", 0), 1)) + "%)"
            if prices[i].get("price") else s + ": unavailable"
            for i, s in enumerate(CRYPTO_WATCHLIST)
        ]
        poly_lines = [
            "[" + str(m.get("yes_price", "?")) + "] " + m.get("question", "")[:70]
            for m in poly[:5] if not m.get("error")
        ]
        reply = "LIVE MARKET\n\n" + "\n".join(price_lines) + "\n\nPOLYMARKET:\n" + ("\n".join(poly_lines) or "unavailable")

    elif tl == "/trades":
        trades = load_trades()[-10:]
        if not trades:
            reply = "No trades yet."
        else:
            lines = ["Last " + str(len(trades)) + " trades:"]
            for tr in reversed(trades):
                mode = "PAPER" if tr.get("paper") else "LIVE"
                lines.append(
                    "[" + mode + "] " + tr.get("side", "").upper() + " " + tr.get("symbol", "") +
                    " @ $" + str(tr.get("price", 0)) + " - " + tr.get("time", "")[:16]
                )
            reply = "\n".join(lines)

    elif tl == "/positions":
        if not trader.positions:
            reply = "No open positions."
        else:
            lines = ["Open positions:"]
            for symbol, pos in trader.positions.items():
                token = symbol.replace("/USDT", "")
                pd = get_price(token)
                current = pd.get("price", 0) or 0
                entry = pos.get("entry_price", 0)
                pnl = ((current - entry) / entry * 100) if entry else 0
                lines.append(symbol + ": entry $" + str(entry) + " | now $" + str(current) + " | PnL " + str(round(pnl, 1)) + "%")
            reply = "\n".join(lines)

    elif tl == "/strategy":
        s = load_strategy()
        reply = (
            "Strategy v" + str(s.get("version", 1)) + "\n"
            "Win rate: " + str(round(s.get("win_rate", 0) * 100, 1)) + "%\n"
            "Total trades: " + str(s.get("total_trades", 0)) + "\n\n"
            "Rules:\n" + s.get("rules", "none") + "\n\n"
            "Notes: " + (", ".join(s.get("notes", [])) or "none")
        )

    elif tl == "/research":
        research = rj(RESEARCH_FILE, [])[:5]
        if not research:
            reply = "No research yet."
        else:
            lines = ["Last " + str(len(research)) + " items:"]
            for r in research:
                lines.append("\n[" + r["time"][:10] + "] " + r["topic"] + "\n" + r["summary"][:200])
            reply = "\n".join(lines)

    elif tl == "/whales":
        whale_log = rj(WHALE_FILE, [])
        if not whale_log:
            reply = "No whale activity recorded yet."
        else:
            latest = whale_log[0]

                "\n".join(latest.get("alerts", [])) +
                "\n\n" + latest.get("analysis", "")[:400]
            )

    elif tl == "/sentiment":
        fg = get_fear_greed()
        news = web_search("crypto sentiment today", max_results=3)
        reply = "Fear & Greed: " + str(fg.get("value")) + " - " + str(fg.get("label")) + "\n\nNews:\n" + news[:400]

    elif tl == "/scan":
        send_telegram("Running full scan now...")
        threading.Thread(target=market_scan_loop, daemon=True).start()
        threading.Thread(target=dex_discovery_loop, daemon=True).start()
        threading.Thread(target=whale_tracker_loop, daemon=True).start()
        reply = "Scans started."

    elif tl.startswith("dex "):
        query = t[4:].strip()
        result = search_dex(query)
        reply = "DexScreener: " + query + "\n\n" + result

    elif tl.startswith("search "):
        query = t[7:].strip()
        result = web_search(query)
        reply = "Search: " + query + "\n\n" + result

    elif tl.startswith("analyze "):
        symbol = t[8:].strip()
        result = web_search(symbol + " price prediction analysis today")
        reply = think("Market research for " + symbol + ":\n" + result + "\n\nDirection, confidence, reasoning, risk.")

    elif tl.startswith("implement "):
        code = t[10:].strip()
        save_pending(code)
        send_telegram("Proposed code:\n\n" + code + "\n\nReply 'yes implement' to apply or 'no implement' to reject.")
        reply = "Proposal saved. Awaiting your approval."

    elif tl == "yes implement":
        pending = load_pending()
        if pending:
            try:
                with open(__file__, "r", encoding="utf-8") as f:
                    current = f.read()
                with open(__file__, "w", encoding="utf-8") as f:
                    f.write(current + "\n\n# === AUTO-IMPLEMENTED " + datetime.now().isoformat() + " ===\n" + pending + "\n")
                save_pending(None)
                reply = "Implemented. Send /restart to reload."
            except Exception as e:

        else:

    elif tl == "no implement":

    elif tl == "/restart":

        os.execl(sys.executable, sys.executable, __file__)

    else:
        market = rj(MARKET_FILE, {})
        ctx = (" [Market context: " + market["summary"] + "]") if market.get("summary") else ""
        reply = think((t or "(empty)") + ctx, history[-20:])
        if "SEARCH:" in reply:
            try:
                query = reply.split("SEARCH:")[1].split("\n")[0].strip()
                results = web_search(query)
                reply = think("Search results for '" + query + "':\n" + results + "\n\nAnswer using these.", history[-20:])
            except Exception:
                pass
        history.append({"role": "user", "content": t or ""})
        history.append({"role": "assistant", "content": reply})
        save_conv(history)

    return reply

def start_listener():
    offset = 0
    print("Nexus listener active...")
    while True:
        try:
            r = requests.get(
                "https://api.telegram.org/bot" + TELEGRAM_BOT_TOKEN + "/getUpdates",
                params={"offset": offset, "timeout": 30}, timeout=35
            )
            updates = r.json().get("result", [])
            for update in updates:
                offset = update["update_id"] + 1
                msg = update.get("message", {})
                text     = msg.get("text", "")
                caption  = msg.get("caption", "")
                photo    = msg.get("photo")
                document = msg.get("document")
                chat_id  = str((msg.get("chat") or {}).get("id", ""))
                if not (text or photo or document):
                    continue
                if chat_id != TELEGRAM_CHAT_ID:
                    continue
                print("[" + datetime.now().strftime("%H:%M:%S") + "] " + (text or "[media]")[:60])
                try:
                    reply = handle_message(text, photo=photo, document=document, caption=caption)
                    send_telegram(reply)
                except Exception as e:
                    send_telegram("Error: " + str(e))
                    print("Handler error: " + str(e))
        except Exception as e:
            print("Listener error: " + str(e))
            time.sleep(5)
        time.sleep(1)
